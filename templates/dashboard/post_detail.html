                            // The aura breathes through sine and time

                            precision mediump float;

                            uniform float u_time;
                            varying vec2 v_uv;

                            void main() {
                                vec2 uv = v_uv * 2.0 - 1.0;

                                float qi = sin(uv.x * 10.0 + u_time) * 0.5;

                                float aura = smoothstep(
                                    0.02,
                                    0.0,
                                    abs(uv.y - qi)
                                );

                                vec3 color = mix(
                                    vec3(0.02, 0.05, 0.03),
                                    vec3(0.85, 0.74, 0.47),
                                    aura
                                );

                                gl_FragColor = vec4(color, 1.0);
                            }
